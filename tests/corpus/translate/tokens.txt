# Token-boundary rewrites that never touch string literals, quoted identifiers or comments.
SELECT NVL(a, b) FROM t => SELECT COALESCE(a, b) FROM t
SELECT SYSDATE FROM DUAL => SELECT CURRENT_TIMESTAMP
SELECT SYSTIMESTAMP FROM DUAL => SELECT CURRENT_TIMESTAMP
SELECT a FROM t1 MINUS SELECT a FROM t2 => SELECT a FROM t1 EXCEPT SELECT a FROM t2
SELECT NVL(NULL, 'SYSDATE'), SYSDATE FROM DUAL MINUS SELECT 'x', SYSDATE FROM DUAL => SELECT COALESCE(NULL, 'SYSDATE'), CURRENT_TIMESTAMP EXCEPT SELECT 'x', CURRENT_TIMESTAMP
# Literals, quoted identifiers and comments are preserved byte for byte.
SELECT 'O''Reilly SYSDATE', "NUMBER", NVL(NULL, 'x') /* MINUS SYSDATE */ FROM DUAL -- NVL => SELECT 'O''Reilly SYSDATE', "NUMBER", COALESCE(NULL, 'x')
# Oracle type names in DDL.
CREATE TABLE oracle_ddl_people (id NUMBER(10) PRIMARY KEY, label VARCHAR2(30), note CLOB, created_at DATE DEFAULT SYSDATE) => CREATE TABLE oracle_ddl_people (id NUMERIC(10) PRIMARY KEY, label VARCHAR(30), note TEXT, created_at DATE DEFAULT CURRENT_TIMESTAMP)
CREATE TABLE t (a NVARCHAR2(10), b BINARY_FLOAT, c BINARY_DOUBLE, d BLOB, e RAW, f LONG) => CREATE TABLE t (a VARCHAR(10), b pgsaci.binary_float, c pgsaci.binary_double, d BYTEA, e BYTEA, f TEXT)
# RAW(n) -> BYTEA (length dropped); VARCHAR2(n CHAR|BYTE) -> VARCHAR(n).
CREATE TABLE t (e RAW(16)) => CREATE TABLE t (e BYTEA)
CREATE TABLE t (a VARCHAR2(5 CHAR), b VARCHAR2(10 BYTE)) => CREATE TABLE t (a VARCHAR(5), b VARCHAR(10))
# Oracle ALTER TABLE spellings.
ALTER TABLE t MODIFY (label VARCHAR2(80), created_at DEFAULT SYSDATE) => ALTER TABLE t ALTER COLUMN label TYPE VARCHAR(80), ALTER COLUMN created_at SET DEFAULT CURRENT_TIMESTAMP
ALTER TABLE t MODIFY (label DEFAULT NULL, required_col NOT NULL, optional_col NULL) => ALTER TABLE t ALTER COLUMN label DROP DEFAULT, ALTER COLUMN required_col SET NOT NULL, ALTER COLUMN optional_col DROP NOT NULL
ALTER TABLE t DROP (obsolete_a, obsolete_b) => ALTER TABLE t DROP COLUMN obsolete_a, DROP COLUMN obsolete_b
ALTER TABLE t SET UNUSED (obsolete_a, obsolete_b) => ALTER TABLE t DROP COLUMN obsolete_a, DROP COLUMN obsolete_b
# PostgreSQL ignores Oracle physical storage tuning.
CREATE TABLE t (id NUMBER) TABLESPACE users PCTFREE 10 INITRANS 2 STORAGE (INITIAL 64K NEXT 1M) LOGGING PARALLEL 4 SEGMENT CREATION IMMEDIATE => CREATE TABLE t (id NUMERIC)
CREATE TABLE t (id NUMBER, payload CLOB) LOB (payload) STORE AS (TABLESPACE users PCTFREE 5) => CREATE TABLE t (id NUMERIC, payload TEXT)
CREATE BITMAP INDEX t_bitmap_ix ON t (id) TABLESPACE users COMPUTE STATISTICS => CREATE INDEX t_bitmap_ix ON t(id)
CREATE TABLE t (TABLESPACE NUMBER, note VARCHAR2(20) DEFAULT 'TABLESPACE stays literal') PCTFREE 5 => CREATE TABLE t (TABLESPACE NUMERIC, note VARCHAR(20) DEFAULT 'TABLESPACE stays literal')
# PostgreSQL has compatible materialized views but no Oracle refresh policy syntax.
CREATE MATERIALIZED VIEW mv BUILD IMMEDIATE REFRESH FAST ON DEMAND AS SELECT id FROM people => CREATE MATERIALIZED VIEW mv AS SELECT id FROM people
# Oracle session settings map to PostgreSQL session state.
ALTER SESSION SET CURRENT_SCHEMA = app_owner => SET search_path TO app_owner, oracle, public
ALTER SESSION SET TIME_ZONE = '-03:00' => SET TIME ZONE '+03:00'
ALTER SESSION SET NLS_DATE_FORMAT = 'YYYY-MM-DD' => SET pgsaci.nls_date_format TO 'YYYY-MM-DD'
ALTER SESSION SET NLS_COMP = LINGUISTIC => SET pgsaci.nls_comp TO LINGUISTIC
ALTER SESSION SET SQL_TRACE = TRUE => RESET application_name
